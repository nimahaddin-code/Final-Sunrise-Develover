// MediaPipe brow shaders of the shared makeup engine. Compiled together with BeautyBlush.metal.txt (appended after it, so the
// shared camera decoding, uniforms and helpers defined there are in scope); see `BeautyBlushShaders`.
// MediaPipe brow. The style's brow is baked once into a brow-space texture: x = u (brow head 0, tail 1), y = offset
// across the brow in brow widths, forehead at the top. A strip laid over the observed brow from this frame's
// MediaPipe landmarks samples it, so the strands stay on the brow as the head moves. Deterministic, never animated.
struct BrowDesign { float4 points[41]; float4 controls; float4 extent; };
// points: u, centre offset, half thickness, t. controls: max alpha, head fade extent, edge feather. extent: u0, u1, wMax.
struct BakeOut { float4 position [[position]]; float2 uv; };
vertex BakeOut browTextureVertex(uint i [[vertex_id]]) {
 float2 p=float2((i<<1)&2,i&2);BakeOut o;o.position=float4(p*2-1,0,1);o.uv=float2(p.x,1-p.y);return o;
}
// r coverage (with max alpha), g strand hair, b design outline for the tutorial guide.
fragment float4 browTextureFragment(BakeOut in [[stage_in]],constant BrowDesign& d [[buffer(0)]]) {
 float2 q=float2(mix(d.extent.x,d.extent.y,in.uv.x),mix(d.extent.z,-d.extent.z,in.uv.y));
 float best=1e8,t=0,v=0;
 for(int i=0;i<40;i++) {
  float4 a=d.points[i],b=d.points[i+1];float2 axis=b.xy-a.xy;
  float f=clamp(dot(q-a.xy,axis)/max(dot(axis,axis),1e-12f),0.f,1.f);
  float4 p=mix(a,b,f);float2 r=q-p.xy;
  float distance=length(r)/max(p.z,1e-6f);
  if(distance<best){best=distance;t=p.w;v=r.y/max(p.z,1e-6f);}
 }
 float feather=clamp(d.controls.z,.03f,.7f);
 float envelope=1-smoothstep(1-feather,1.f,best);
 float head=smoothstep(0.f,max(d.controls.y,.02f),t);
 float tail=1-smoothstep(.97f,1.f,t);
 float base=envelope*head*tail;
 // Upright heads gradually sweep outward. Irregular, subpixel-soft strand spacing avoids a solid sticker.
 float phase=t*78.f-v*mix(.28f,1.8f,t)+.15f*sin(t*93.f)+.09f*sin(v*8.f+t*47.f);
 float width=max(fwidth(phase),.055f);
 float strand=1-smoothstep(.08f,.08f+width,abs(fract(phase)-.5f));
 float outline=(1-smoothstep(0.f,.09f,abs(best-1)))*step(.02f,t)*step(t,.98f);
 return float4(base*d.controls.x,base*strand,outline,0);
}
struct BrowVertex { float4 position; float4 brow; float4 probes; float4 skin; float4 conceal; };
// position: clip. brow: u, offset in brow widths, observed half thickness, side fade. probes: camera coordinates a
// little above and below (for occlusion). skin: camera-coordinate blur radius (xy) for the detail layer. conceal:
// lower edge of the hidden-brow area (core, end, in brow widths, bounded by the eyelid row) and side (0 right, 1 left).
struct BrowOut { float4 position [[position]]; float4 brow; float4 probes; float4 skin; float4 conceal; };
vertex BrowOut browStripVertex(uint i [[vertex_id]],constant BrowVertex* v [[buffer(0)]]) {
 BrowOut o;o.position=v[i].position;o.brow=v[i].brow;o.probes=v[i].probes;o.skin=v[i].skin;o.conceal=v[i].conceal;return o;
}
// ---- Hiding the natural brow ----
// Each brow is unwrapped into its own brow-space texture (u along the brow, w across it), the hair area is filled
// from the clean skin around it by push-pull over the mip pyramid, given forehead texture, smoothed over time, and
// laid back onto the face by the brow strip. An inpainting model's result can replace the fill (see BrowInpainter).
float lum709(float3 c) { return dot(c,float3(.2126,.7152,.0722)); }
// Area that hides the natural brow, 0...1 (see concealUpper/concealLower). u, w and the half thickness h are in brow
// widths; coreDown/endDown are the lower edge, bounded by the eyelid row.
float browHide(float u,float w,float h,float coreDown,float endDown) {
 bool up=w>=0;
 float across=1-smoothstep(up ? h*1.15f+.02f:coreDown,up ? h*2.2f+.10f:endDown,abs(w));
 // Long fades at the head and tail: a hard end by the nose bridge read as a patch.
 // MediaPipe's tail row ends short of the real brow tail, so the tail fade starts later and runs further than the head's.
 float along=smoothstep(-.12f,.06f,u)*(1-smoothstep(1.0f,1.28f,u));
 float x=across*along;
 return x*x*x*(x*(x*6-15)+10);
}
struct UnwrapVertex { float4 brow; float4 raw; float4 conceal; };
// brow: u, w, observed half thickness, side. raw: camera coordinates (xy) and the model's face-crop clip position (zw).
// conceal: lower core and end of the hidden area.
struct UnwrapOut { float4 position [[position]]; float4 brow; float4 raw; float2 conceal; };
// space: u0, u1, w at the bottom and at the top of the brow-space texture.
vertex UnwrapOut browUnwrapVertex(uint i [[vertex_id]],constant UnwrapVertex* v [[buffer(0)]],constant float4& space [[buffer(1)]]) {
 UnwrapOut o;float4 b=v[i].brow;
 float x=(b.x-space.x)/(space.y-space.x),y=(space.w-b.y)/(space.w-space.z);
 o.position=float4(x*2-1,1-y*2,0,1);o.brow=b;o.raw=v[i].raw;o.conceal=v[i].conceal.xy;return o;
}
// Camera colour in brow space: premultiplied by how much it is known skin from the landmarks alone (0 in the area
// between and around MediaPipe's brow rows), and unmasked for hair detection and texture.
struct UnwrapTargets { float4 known [[color(0)]]; float4 raw [[color(1)]]; };
fragment UnwrapTargets browUnwrapFragment(UnwrapOut in [[stage_in]],constant U& u [[buffer(0)]],texture2d<float> y [[texture(0)]],texture2d<float> cb [[texture(1)]]) {
 float3 c=cameraPixel(in.raw.xy,u,y,cb);
 float hole=saturate(browHide(in.brow.x,in.brow.y,in.brow.z,in.conceal.x,in.conceal.y)*1.6f);
 float known=(1-hole)*(1-hole);
 UnwrapTargets o;o.known=float4(c*known,known);o.raw=float4(c,1);return o;
}
// The actual brow edge: MediaPipe's rows sit inside the hair, so pixels around them that are clearly darker than the
// skin estimated without them are brow hair too. Searched a little further than the landmark area (more above,
// never below the eyelid bound), grown by a few pixels, and added to the hidden area. Writes the known-skin layer for
// the final fill and the hidden-area mask the brow strip and the model use.
struct HairTargets { float4 known [[color(0)]]; float hide [[color(1)]]; };
fragment HairTargets browHairFragment(UnwrapOut in [[stage_in]],constant float4& space [[buffer(0)]],texture2d_array<float> raw [[texture(0)]],texture2d_array<float> estimate [[texture(1)]]) {
 constexpr sampler s(coord::normalized,address::clamp_to_edge,filter::linear,mip_filter::linear);
 uint slice=uint(in.brow.w+.5f);
 float2 uv=float2((in.brow.x-space.x)/(space.y-space.x),(space.w-in.brow.y)/(space.w-space.z));
 float h=in.brow.z;
 bool up=in.brow.y>=0;
 float reach=up ? h*2.6f+.14f:in.conceal.y;
 float region=(1-smoothstep(reach*.8f,reach,abs(in.brow.y)))*smoothstep(-.26f,-.08f,in.brow.x)*(1-smoothstep(1.12f,1.3f,in.brow.x));
 float skin=max(lum709(estimate.sample(s,uv,slice).rgb),1e-4f);
 float2 px=float2(1.f/raw.get_width(),1.f/raw.get_height());
 float hair=0;
 for(int i=0;i<5;i++) {
  float2 o=i==0 ? float2(0):float2(i==1 ? 2.f:(i==2 ? -2.f:0.f),i==3 ? 2.f:(i==4 ? -2.f:0.f))*px;
  hair=max(hair,smoothstep(.035f,.11f,1-lum709(raw.sample(s,uv+o,slice,level(.5f)).rgb)/skin));
 }
 float landmarks=browHide(in.brow.x,in.brow.y,in.brow.z,in.conceal.x,in.conceal.y);
 float hide=max(landmarks,hair*region);
 float hole=saturate(hide*1.6f),known=(1-hole)*(1-hole);
 float3 c=raw.sample(s,uv,slice,level(0.f)).rgb;
 HairTargets o;o.known=float4(c*known,known);o.hide=hide;return o;
}
struct FillParams { float4 a; float4 space; float4 options; };
// a: slice, temporal blend (1 = this frame only), model weight, mip level count. options: forehead texture amount,
// finest level used (0 for the fill; coarser for the skin estimate, so hair beside the landmarks does not count as skin).
// Push-pull: from the coarsest level down, each level's known colour replaces the coarser estimate as far as it is
// known, so the hole gets a smooth fill that meets the skin on every side. Full-resolution texture from a band of
// forehead above the brow is laid on top (the model's 256-pixel face crop alone looks hazy once enlarged); the result
// is blended with the previous frame.
fragment float4 browFillFragment(BakeOut in [[stage_in]],constant FillParams& p [[buffer(0)]],texture2d_array<float> unwrap [[texture(0)]],texture2d_array<float> previous [[texture(1)]],texture2d_array<float> model [[texture(2)]],texture2d_array<float> raw [[texture(3)]]) {
 constexpr sampler s(coord::normalized,address::clamp_to_edge,filter::linear,mip_filter::linear);
 uint slice=uint(p.a.x+.5f);int levels=int(p.a.w+.5f);
 float3 color=0;bool started=false;
 for(int l=levels-1;l>=int(p.options.y+.5f);l--) {
  float4 t=unwrap.sample(s,in.uv,slice,level(float(l)));
  if(t.a<1e-5f) continue;
  float3 c=t.rgb/t.a;
  color=started ? mix(color,c,saturate(t.a*4)):c;started=true;
 }
 if(p.a.z>0) color=mix(color,model.sample(s,in.uv,slice).rgb,p.a.z);
 // Forehead texture: the same column, a repeating 0.18-brow-width band starting 0.45 above the centerline. Dark
 // detail (stray hair) counts less than light.
 float w=mix(p.space.w,p.space.z,in.uv.y);
 float wSource=.45f+fract((w-p.space.z)/.18f)*.18f;
 float2 source=float2(in.uv.x,(p.space.w-wSource)/(p.space.w-p.space.z));
 float detail=lum709(raw.sample(s,source,slice,level(0.f)).rgb)/max(lum709(raw.sample(s,source,slice,level(2.5f)).rgb),1e-4f)-1;
 detail=clamp(detail*(detail<0 ? .6f:1.f),-.08f,.12f);
 color*=1+detail*p.options.x;
 float3 before=previous.sample(s,in.uv,slice).rgb;
 return float4(mix(before,color,p.a.y),1);
}
// Face crop for the inpainting model: upright, sRGB-encoded.
fragment float4 browCropFragment(BakeOut in [[stage_in]],constant U& u [[buffer(0)]],constant float4x4& cropToRaw [[buffer(1)]],texture2d<float> y [[texture(0)]],texture2d<float> cb [[texture(1)]]) {
 float3 c=saturate(cameraPixel((cropToRaw*float4(in.uv,0,1)).xy,u,y,cb));
 return float4(select(1.055f*pow(c,1/2.4f)-.055f,c*12.92f,c<=.0031308f),1);
}
// The hidden area in the face crop (1 = fill), for the model's mask, from this frame's hide mask.
vertex UnwrapOut browCropMaskVertex(uint i [[vertex_id]],constant UnwrapVertex* v [[buffer(0)]]) {
 UnwrapOut o;o.position=float4(v[i].raw.zw,0,1);o.brow=v[i].brow;o.raw=v[i].raw;o.conceal=v[i].conceal.xy;return o;
}
fragment float4 browCropMaskFragment(UnwrapOut in [[stage_in]],constant float4& space [[buffer(0)]],texture2d_array<float> hide [[texture(0)]]) {
 constexpr sampler s(coord::normalized,address::clamp_to_edge,filter::linear,mip_filter::linear);
 float2 uv=float2((in.brow.x-space.x)/(space.y-space.x),(space.w-in.brow.y)/(space.w-space.z));
 return float4(step(.02f,hide.sample(s,uv,uint(in.brow.w+.5f),level(1.f)).r),0,0,1);
}
// The model's result (sRGB face crop) back into brow space, using the geometry of the frame it was asked about.
fragment float4 browModelFragment(UnwrapOut in [[stage_in]],texture2d<float> crop [[texture(0)]]) {
 constexpr sampler s(coord::normalized,address::clamp_to_edge,filter::linear);
 float3 c=crop.sample(s,float2(in.raw.z*.5f+.5f,.5f-in.raw.w*.5f)).rgb;
 return float4(select(pow((c+.055f)/1.055f,float3(2.4f)),c/12.92f,c<=.04045f),1);
}
fragment float4 browStripFragment(BrowOut in [[stage_in]],constant U& u [[buffer(0)]],constant float4& extent [[buffer(1)]],constant float4& space [[buffer(2)]],texture2d<float> y [[texture(0)]],texture2d<float> cb [[texture(1)]],texture2d<float> brow [[texture(2)]],texture2d<float> obstruction [[texture(3)]],texture2d_array<float> fill [[texture(4)]],texture2d_array<float> hideMask [[texture(5)]]) {
 constexpr sampler s(coord::normalized,address::clamp_to_zero,filter::linear,mip_filter::linear,max_anisotropy(4));
 constexpr sampler m(coord::normalized,address::clamp_to_edge,filter::linear);
 float2 raw=(u.rawFromClip*float4(in.position.xy*u.tracking.zw,0,1)).xy;
 float3 skin=cameraPixel(raw,u,y,cb);
 // skin.z: an overlay picture is drawn in chord space, so its row is offset by the centerline's rise above the chord (0 for shapes).
 // An overlay picture is sampled a little blurrier: fine strands crawling with the tracking read as glitter.
 float4 packed=brow.sample(s,float2((in.brow.x-extent.x)/(extent.y-extent.x),(extent.z-(in.brow.y+in.skin.z))/(2*extent.z)),bias(u.debug.w>.5f ? .5f:0.f));
 float coverage=packed.r;
 float occ=obstruction.sample(m,(u.maskFromRaw*float4(raw,0,1)).xy).r;
 // debug.y: hide the natural brow. debug.z: draw the brow shape.
 bool conceal=u.debug.y>.5f;
 if(coverage>.001f || conceal) {
  // The face-skin segmenter often rejects brow hair itself. Treat the brow as occluded only when the skin above or
  // below is too (a hand or a fringe), not when only the hair is rejected.
  float above=obstruction.sample(m,(u.maskFromRaw*float4(in.probes.xy,0,1)).xy).r;
  float below=obstruction.sample(m,(u.maskFromRaw*float4(in.probes.zw,0,1)).xy).r;
  occ=min(occ,max(above,below));
 }
 float lum=max(dot(skin,float3(.2126,.7152,.0722)),.0001f);
 // Hidden natural brow: the brow-space fill (push-pull or model) where the area hides it, soft-edged.
 // The mask includes the detected brow edge; sampled one mip level down so its edge stays soft.
 constexpr sampler fs(coord::normalized,address::clamp_to_edge,filter::linear,mip_filter::linear);
 float hide=0;
 float3 hidden=skin;
 float original=0;
 // Fill and hide-mask textures are only needed when natural-brow concealment is
 // enabled. Overlay-only frames avoid two array-texture samples per fragment.
 if(conceal) {
  float2 fuv=float2((in.brow.x-space.x)/(space.y-space.x),(space.w-in.brow.y)/(space.w-space.z));
  uint fslice=uint(in.conceal.z+.5f);
  hide=hideMask.sample(fs,fuv,fslice,level(1.f)).r;
  hidden=fill.sample(fs,fuv,fslice).rgb;
  // Original brow pixels (debug view): inside that area and darker than the fill.
  original=hide*smoothstep(.03f,.16f,1-lum/max(lum709(hidden),1e-4f));
 }
 // Hide the natural brow. Hands and fringes stay in front. The lower edge
 // sits against the eyelid, where a full-strength forehead fill reads as a
 // dark strip on warm skin; ease that replacement toward the live skin tone.
 float concealAmount=hide*(1-occ)*in.brow.w;
 if(in.brow.y<0) concealAmount*=.52f;
 float3 base=conceal ? mix(skin,hidden,concealAmount):skin;
 // Brow shape, overlay only: output = alpha * hair + (1 - alpha) * base. Alpha is zero outside the designed brow, so
 // every other pixel is unchanged. Between strands the skin stays visible.
 float density=mix(.35f,1.f,saturate(packed.g/max(coverage,.001f)));
 float browAlpha=saturate(coverage*density*u.shade.w*u.tracking.x*(1-occ)*in.brow.w)*step(.5f,u.debug.z);
 // debug.w: an overlay picture; its strand detail (packed.a, 1 = average) rides on the shade.
 // An overlay is lit by the smooth skin under it (the fill where the natural brow is hidden), not by the camera pixel,
 // whose natural hair and noise would show through the picture as shimmer.
 float lit=u.debug.w>.5f && conceal ? max(lum709(base),1e-4f):lum;
 // The strand detail also adds a little light, so strands show on a black shade too.
 float strand=u.debug.w>.5f ? packed.a-1.f:0.f;
 float3 hairColor=clamp(u.shade.rgb*clamp(sqrt(lit/.25f),.55f,1.4f)*(1+strand)+strand*.04f,0.f,1.f);
 float3 result=mix(base,hairColor,browAlpha);
 if(u.tracking.y>.5) result=mix(result,float3(.95,.72,.80),.7f*packed.b*u.tracking.x*in.brow.w);
 // Debug views: 1 original, 2 landmarks (observed contour), 3 fit contours, 4 alpha mask, 5 texture only, 6 original brow mask.
 int view=int(u.debug.x+.5f);
 if(view==1) result=skin;
 else if(view==2) result=mix(skin,float3(1,.9,.1),1-smoothstep(0.f,.012f,abs(abs(in.brow.y)-in.brow.z)));
 else if(view==3) result=mix(mix(skin,float3(.1,.9,1),step(.5f,packed.b)),float3(1,.9,.1),1-smoothstep(0.f,.01f,abs(in.brow.y)));
 else if(view==4) result=float3(browAlpha);
 else if(view==5) result=mix(float3(.5),hairColor,browAlpha);
 else if(view==6) result=float3(original);
 return float4(result,1);
}
// Tutorial nose-eye mapping lines and landmark debug dots, alpha blended over the frame.
struct MarkVertex { float4 position; float4 color; };
struct MarkOut { float4 position [[position]]; float4 color; };
vertex MarkOut browMarkVertex(uint i [[vertex_id]],constant MarkVertex* v [[buffer(0)]]) { MarkOut o;o.position=v[i].position;o.color=v[i].color;return o; }
fragment float4 browMarkFragment(MarkOut in [[stage_in]]) { return in.color; }
// ---- 3D hair brow ----
// Each hair is a ribbon of the brow mesh, placed on the face by ARKit's pose (clipFromModel). A hair is thinner than a pixel, so
// it is widened on screen to a minimum width and its coverage lowered by the same ratio: the brow keeps its density and the strands
// do not flicker in and out as they move across pixel centres.
struct HairVertex { float4 a; float4 b; float4 extra; };
// a: end A (xyz, model metres) and width. b: end B and tone (1 = the shade). extra: end (0 A, 1 B), side (-1, 1).
struct HairUniforms { float4x4 clipFromModel; float4 view; float4 look; float4 probe; };
// view: drawable width, height (pixels), pixels per metre at w = 1, model-to-face scale. look: light, minimum width (pixels), tone's added light.
// probe: offset in the raw image (normalized) used to look above and below a strand when asking whether a hand covers it.
struct HairOut { float4 position [[position]]; float tone; float cover; };
vertex HairOut browRibbonVertex(uint i [[vertex_id]],constant HairVertex* v [[buffer(0)]],constant HairUniforms& h [[buffer(1)]]) {
 HairVertex r=v[i];HairOut o;
 float4 ca=h.clipFromModel*float4(r.a.xyz,1),cb=h.clipFromModel*float4(r.b.xyz,1);
 if(ca.w<=1e-4f||cb.w<=1e-4f) {o.position=float4(2,2,2,1);o.tone=1;o.cover=0;return o;}
 float4 c=r.extra.x>.5f ? cb:ca;
 float2 d=(cb.xy/cb.w-ca.xy/ca.w)*h.view.xy*.5f;
 float len=length(d);d=len>1e-4f ? d/len:float2(1,0);
 float truePx=r.a.w*h.view.w*h.view.z/c.w;
 float px=max(truePx,h.look.y);
 float2 offset=(float2(-d.y,d.x)*r.extra.y+d*(r.extra.x>.5f ? 1.f:-1.f))*px*.5f/(h.view.xy*.5f);
 o.position=float4(c.xy+offset*c.w,c.z,c.w);
 o.tone=r.b.w;o.cover=saturate(truePx/px);
 return o;
}
fragment float4 browRibbonFragment(HairOut in [[stage_in]],constant U& u [[buffer(0)]],constant HairUniforms& h [[buffer(1)]],texture2d<float> obstruction [[texture(3)]]) {
 constexpr sampler m(coord::normalized,address::clamp_to_edge,filter::linear);
 float2 raw=(u.rawFromClip*float4(in.position.xy*u.tracking.zw,0,1)).xy;
 float occ=obstruction.sample(m,(u.maskFromRaw*float4(raw,0,1)).xy).r;
 // The face-skin segmenter often rejects brow hair itself: a strand counts as covered only when the skin beside it is too.
 float above=obstruction.sample(m,(u.maskFromRaw*float4(raw+h.probe.xy,0,1)).xy).r;
 float below=obstruction.sample(m,(u.maskFromRaw*float4(raw-h.probe.xy,0,1)).xy).r;
 occ=min(occ,max(above,below));
 float3 color=clamp(u.shade.rgb*h.look.x*in.tone+(in.tone-1)*h.look.z,0.f,1.f);
 return float4(color,saturate(in.cover*u.shade.w*u.tracking.x*(1-occ)));
}

// -Nimah: One camera/mesh timestamp, native face UVs, linear-light matte pigment. No skin smoothing.
#include <metal_stdlib>
using namespace metal;
struct V { float4 position; float4 rest; float4 uv; };
struct U { float4x4 mvp; float4x4 rawFromClip; float4x4 maskFromRaw;
 float4 shade; float4 controls; float4 tracking; float4 camera; float4 stretch; float4 debug; float4 blush; };
struct O { float4 position [[position]]; float2 uv; float3 local; float3 rest; };
vertex O blushFaceVertex(uint i [[vertex_id]],constant V* v [[buffer(0)]],constant U& u [[buffer(1)]]) {
 O o;o.position=u.mvp*v[i].position;o.uv=v[i].uv.xy;o.local=v[i].position.xyz;o.rest=v[i].rest.xyz;return o;
}
vertex O blushAtlasVertex(uint i [[vertex_id]],constant V* v [[buffer(0)]]) {
 O o;o.position=float4(v[i].uv.x*2-1,1-v[i].uv.y*2,0,1);o.uv=v[i].uv.xy;o.local=v[i].rest.xyz;o.rest=o.local;return o;
}
struct Field { float4 ellipse; float4 shape; float4 limits; };
struct Bake { Field fields[2];Field bridge;float4 paint;float4 options;float4 mouth; };
fragment float4 blushAtlasFragment(O in [[stage_in]],constant Bake& b [[buffer(0)]],texture2d<float> brush [[texture(0)]]) {
 constexpr sampler s(coord::normalized,address::clamp_to_zero,filter::linear);
 float3 masks=0;
 for(int i=0;i<3;i++) {
  Field f=i<2 ? b.fields[i]:b.bridge;
  if(f.ellipse.z<=0 || f.ellipse.w<=0) continue;float2 d=in.local.xy-f.ellipse.xy;d.x*=f.shape.x;
  float co=cos(f.shape.y),si=sin(f.shape.y);float2 p=float2(d.x*co+d.y*si,-d.x*si+d.y*co)/f.ellipse.zw;
  p.y -= b.paint.x*(p.x*p.x-.25f)*(i<2 ? 1.f:0.f);
  float width=clamp(b.options.z,.2f,1.f);
  float density=smoothstep(.5-width*.5,.5+width*.5,brush.sample(s,float2(p.x*.5+.5,.5-p.y*.5)).r);
  float a=pow(density,max(f.shape.w,.001f));
  if(b.paint.z>.5) {
   // Landmark-fitted wash: flat-topped, zero slope at the rim, so no edge is ever visible.
   a=1-smoothstep(0.f,1.f,length(p));
  } else if(b.paint.y>.5) {
   float r2=dot(p,p);
   a=pow(max(0.f,1-r2),max(f.shape.w,1.f))*exp(-.8f*r2);
  } else if(i<2) {a=mix(a,a*a*(3-2*a),b.options[i]);}
  float w=max(f.limits.w,.00001);
  a*=smoothstep(0.f,w,in.local.x*f.shape.x-f.limits.x);
  a*=smoothstep(0.f,w,f.limits.y-in.local.x*f.shape.x);
  a*=smoothstep(0.f,w,f.limits.z-in.local.y);
  if(b.mouth.z>0 && b.mouth.w>0) {
   float lipDistance=length((in.local.xy-b.mouth.xy)/b.mouth.zw);
   a*=smoothstep(1.f,1.2f,lipDistance);
  }
  masks[i]=a*f.shape.z;
 }
 return float4(masks,1);
}
vertex O blushBackgroundVertex(uint i [[vertex_id]]) {
 float2 p=float2((i<<1)&2,i&2);O o;o.position=float4(p*2-1,1,1);o.uv=float2(p.x,1-p.y);o.local=0;o.rest=0;return o;
}
// The camera tone ARSCNView shows (the eyeliner tab, Try On switched off), measured on iPhone 17 from paired screenshots
// on 2026-10-30: encoded camera value at 0, 0.1 ... 1 -> encoded value on screen. Neither standard curve matched (sRGB
// decode: up to 13/255 too dark in the mid-tones; BT.709 decode: 11/255 too light in the shadows, "hazy"); this table
// left under 1.5/255 of systematic error per brightness band on blocks not used to fit it. Linear between knots, then
// the inverse of the drawable's sRGB encoding, so the screen shows exactly the table's value.
constant float cameraTone[11]={0.f,.1228f,.2301f,.3507f,.4543f,.544f,.6322f,.7219f,.8087f,.8916f,.9842f};
float toneOf(float x) {
 float t=clamp(x,0.f,1.f)*10;int i=min(int(t),9);
 return mix(cameraTone[i],cameraTone[i+1],t-float(i));
}
float3 decodeCamera(float3 x) {
 x=float3(toneOf(x.r),toneOf(x.g),toneOf(x.b));
 return select(x/12.92,pow((x+.055)/1.055,float3(2.4)),x>.04045);
}
float3 cameraPixel(float2 raw,constant U& u,texture2d<float> y,texture2d<float> cb) {
 constexpr sampler s(coord::normalized,address::clamp_to_edge,filter::linear);
 float l=y.sample(s,raw).r;float2 c=cb.sample(s,raw).rg-.5;
 if(u.camera.y<.5){l=(l-16.f/255)*255.f/219;c*=255.f/224;}
 float3 rgb=u.camera.x>.5 ? float3(l+1.5748*c.y,l-.187324*c.x-.468124*c.y,l+1.8556*c.x)
 : float3(l+1.402*c.y,l-.344136*c.x-.714136*c.y,l+1.772*c.x);
 return decodeCamera(rgb);
}
fragment float4 blushBackgroundFragment(O in [[stage_in]],constant U& u [[buffer(0)]],texture2d<float> y [[texture(0)]],texture2d<float> cb [[texture(1)]]) {
 // Brow debug: alpha mask and original-brow mask over black, texture only over neutral grey.
 if(u.camera.w>.5 && u.debug.x>3.5) return float4(float3(abs(u.debug.x-5)<.5f ? .5f:0.f),1);
 float2 raw=(u.rawFromClip*float4(in.uv,0,1)).xy;return float4(cameraPixel(raw,u,y,cb),1);
}
fragment float4 blushDepthFragment(O in [[stage_in]]) { return 0; }
float lumOf(float3 c) {return dot(c,float3(.2126,.7152,.0722));}
// Pulls a colour back into 0...1 along the line to its own luminance: hue and luminance stay, only chroma shrinks.
// Per-channel clamping instead shifts hue and drops luminance on bright skin.
float3 intoGamut(float3 c) {
 float l=clamp(lumOf(c),0.f,1.f);float3 d=c-l;float k=1;
 for(int i=0;i<3;i++) {if(d[i]>1e-5f) k=min(k,(1-l)/d[i]);else if(d[i]< -1e-5f) k=min(k,l/(-d[i]));}
 return l+d*max(k,0.f);
}
// Low-frequency skin around `raw`: the centre plus two rings, `u.blush.z` and half of it, in camera pixels.
// Pores, fine texture and small highlights are what `skin` has and this does not.
float3 cameraBase(float2 raw,float3 centre,constant U& u,texture2d<float> y,texture2d<float> cb) {
 float2 px=max(u.blush.z,1.f)/float2(y.get_width(),y.get_height());
 float3 sum=centre*2;
 for(int i=0;i<8;i++) {float a=float(i)*.785398f;sum+=cameraPixel(raw+px*float2(cos(a),sin(a)),u,y,cb);}
 for(int i=0;i<4;i++) {float a=float(i)*1.570796f+.392699f;sum+=cameraPixel(raw+.5f*px*float2(cos(a),sin(a)),u,y,cb);}
 return sum/14;
}
// The pigment colour over `base`: an absorbing layer (controls.y, Beer-Lambert) mixed with the shade's own body at
// the skin's luminance (controls.z how much body, controls.w body vs absorption). Linear light throughout.
float3 pigmentTarget(float3 base,constant U& u) {
 float lb=max(lumOf(base),1e-4f);
 float3 shade=u.shade.rgb;float sl=max(lumOf(shade),.02f);
 float3 body=intoGamut(shade*(lb/sl));
 float3 shadeN=shade/max(max(shade.r,shade.g),max(shade.b,.001f));
 float3 filtered=base*exp(-u.controls.y*(1-shadeN));
 float3 t=mix(filtered,mix(base,body,u.controls.z),u.controls.w);
 // Blush tints the skin; it should not darken it. The absorbing layer alone took ~15% of the brightness, read on device
 // as a darker face. Give back all but 5% of the skin's own luminance (intoGamut later keeps the hue in range).
 return t*min(lb*.95f/max(lumOf(t),1e-4f),4.f);
}
// Antialiased isoline of `m` at `level`, about 1.5 px wide.
float isoline(float m,float level) {float w=max(fwidth(m),1e-5f);return 1-smoothstep(0.f,1.5f*w,abs(m-level));}
// u.blush: x blend (0 skin-preserving pigment, 1 previous blend, 2 linear alpha, 3 multiply, 4 soft light),
// y diagnostic (0 final, 1 neutral geometry, 2 UV checker, 3 pigment mask, 4 mask boundaries, 5 visibility), z detail radius.
fragment float4 blushPigmentFragment(O in [[stage_in]],constant U& u [[buffer(0)]],texture2d<float> y [[texture(0)]],texture2d<float> cb [[texture(1)]],texture2d<float> atlas [[texture(2)]],texture2d<float> obstruction [[texture(3)]]) {
 constexpr sampler s(coord::normalized,address::clamp_to_edge,filter::linear,mip_filter::linear,max_anisotropy(4));
 float2 viewUV=in.position.xy*u.tracking.zw;
 float2 raw=(u.rawFromClip*float4(viewUV,0,1)).xy;
 float3 skin=cameraPixel(raw,u,y,cb);
 float2 maskUV=(u.maskFromRaw*float4(raw,0,1)).xy;
 float occ=obstruction.sample(s,maskUV).r;
 // Do not let small confidence noise modulate clear facial skin every frame.
 if(u.camera.w<.5) occ=smoothstep(.12f,.65f,occ);
 float4 atlasValue=atlas.sample(s,in.uv);
 float2 masks=atlasValue.rg;
 float pigmentMask=max(masks.x,masks.y);
 if(u.camera.w<.5) pigmentMask=max(pigmentMask,atlasValue.b);
 float area=length(cross(dfdx(in.local),dfdy(in.local)))/max(length(cross(dfdx(in.rest),dfdy(in.rest))),1e-10f);
 float stretchFade=smoothstep(u.stretch.x,u.stretch.y,area)*(1-smoothstep(u.stretch.z,u.stretch.w,area));
 float visibility=u.tracking.x*(1-occ)*stretchFade;
 float alpha=saturate(pigmentMask*u.shade.w*u.controls.x*visibility);
 int diagnostic=int(u.blush.y+.5f);
 if(diagnostic==1) return float4(mix(skin,float3(0,.75,.85),saturate(pigmentMask*1.8f)*visibility),1);
 if(diagnostic==2) {float2 c=floor(in.uv*24);return float4(mix(skin,float3(fmod(c.x+c.y,2.f)),.45f),1);}
 if(diagnostic==3) return float4(float3(saturate(pigmentMask*2)),1);
 if(diagnostic==4) {
  float3 r=skin;
  r=mix(r,float3(.1,.9,1),isoline(pigmentMask,.03f));r=mix(r,float3(1,.85,.1),isoline(pigmentMask,.15f));r=mix(r,float3(1,.2,.4),isoline(pigmentMask,.3f));
  return float4(r,1);
 }
 if(diagnostic==5) return float4(mix(float3(lumOf(skin))*.4f,float3(occ,stretchFade,u.tracking.x),.75f),1);
 int mode=int(u.blush.x+.5f);
 float3 result;
 if(mode==1) {
  // Previous blend, kept for comparison on the same frame: recolours skin per pixel and clamps channels.
  float lum=max(lumOf(skin),.0001f);float shadeLum=max(lumOf(u.shade.rgb),.02f);
  float3 tint=u.shade.rgb*(lum/shadeLum);
  float3 absorption=skin*exp(-u.controls.y*(1-u.shade.rgb/max(max(u.shade.r,u.shade.g),max(u.shade.b,.001f))));
  result=mix(skin,clamp(mix(absorption,mix(skin,tint,u.controls.z),u.controls.w),0.f,1.f),alpha);
 } else if(mode==2) {
  result=mix(skin,intoGamut(u.shade.rgb),alpha);
 } else if(mode==3) {
  result=skin*mix(float3(1),u.shade.rgb/max(max(u.shade.r,u.shade.g),max(u.shade.b,.001f)),alpha);
 } else if(mode==4) {
  float3 b=u.shade.rgb;result=mix(skin,(1-2*b)*skin*skin+2*b*skin,alpha);
 } else {
  // Skin-preserving pigment: colour the low-frequency skin, then put back this pixel's own detail -- its luminance
  // ratio to the base (pores, fine texture, highlights, cheek shading) and its chroma residual (freckles, redness).
  // With alpha 0 this is exactly `skin`. Strong local highlights take less pigment so the shine stays near white.
  float3 base=cameraBase(raw,skin,u,y,cb);
  float dl=lumOf(skin)/max(lumOf(base),1e-4f);
  float a=alpha*(1-.6f*smoothstep(1.15f,1.6f,dl));
  // Full pigment brought into gamut, then mixed by alpha: with alpha 0 the camera pixel passes through untouched.
  float3 pigmented=intoGamut(skin+(pigmentTarget(base,u)-base)*dl);
  result=mix(skin,pigmented,a);
 }
 // Optional tutorial contour, entirely separate from pigment alpha; antialiased instead of a hard 0.02-0.045 band.
 if(u.tracking.y>.5) result=mix(result,float3(.95,.72,.80),.7f*isoline(pigmentMask,.03f));
 return float4(result,1);
}
